KEY FEATURES:

================================================================================
COMMON UI PATTERNS
================================================================================

1. REACTIVE SUBSCRIPTION PATTERN
   void Start() => SomeEvent.Subscribe(HandleEvent);
   void OnDestroy() => SomeEvent.Unsubscribe(HandleEvent);

2. PANEL TOGGLE ARRAY PATTERN
   [SerializeField] GameObject[] m_Panels;
   public void TogglePanel(int id)
   {
       for (int i = 0; i < m_Panels.Length; i++)
           m_Panels[i].SetActive(i == id);
   }

3. TMP TEXT WITH DYNAMIC SIZING
   text.rectTransform.sizeDelta = new Vector2(
       text.preferredWidth * scale, 
       text.rectTransform.sizeDelta.y
   );

4. FILL IMAGE FOR PROGRESS/AMPLITUDE
   m_FillImage.fillAmount = normalizedValue;

5. SPRITE SWAP FOR STATE
   m_Icon.sprite = isActive ? m_OnSprite : m_OffSprite;
